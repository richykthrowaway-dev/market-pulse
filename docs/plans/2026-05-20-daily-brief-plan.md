# Daily Market Brief — Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add an AI-generated daily market brief card to the top of the Dashboard — generated once at 6am ET via pg_cron, cached in Supabase, served to all users with zero per-user cost.

**Architecture:** A `generate-daily-brief` Supabase edge function is triggered by pg_cron at 6:00 AM ET (10:00 UTC) on weekdays. It fetches EODHD breadth data, Yahoo Finance index/sector quotes, and Finnhub headlines in parallel, computes derived metrics, calls the Claude API once for prose fields, then upserts one row to the `daily_briefs` table. The frontend reads the latest row via React Query (staleTime 1h) and renders `DailyBriefCard` above the Dashboard stats row.

**Tech Stack:** Deno (Supabase edge function), Claude API (claude-3-5-haiku-20241022), PostgreSQL + pg_cron, React + TanStack Query v5, Tailwind CSS design tokens, TypeScript

```sql
-- Daily market brief cache
-- One row per trading day. Generated once at 6am ET by the generate-daily-brief
-- edge function. Served read-only to all frontend users via RLS policy.

CREATE TABLE IF NOT EXISTS daily_briefs (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  date            DATE NOT NULL UNIQUE,          -- "2026-05-20" — one row per day, enforced at DB layer
  headline        TEXT NOT NULL,                 -- one-line summary for collapsed card state
  market_regime   TEXT NOT NULL,                 -- trending_up | range_bound | volatile | risk_off
  sentiment_score INTEGER NOT NULL,              -- 0–100 composite (breadth formula)
  vix             NUMERIC(6,2),                  -- VIX closing value (nullable — may be missing)
  summary         TEXT NOT NULL DEFAULT '',      -- AI 3–4 sentence narrative ('' if Claude failed)
  sectors         JSONB NOT NULL DEFAULT '[]',   -- [{ symbol, name, pct_change }] sorted desc
  fear_greed      JSONB NOT NULL DEFAULT '{}',   -- { score, label, vix, interpretation }
  current_events  TEXT NOT NULL DEFAULT '',      -- AI macro/geopolitical blurb
  calendar        JSONB NOT NULL DEFAULT '[]',   -- [{ time, name, estimate, prior }]
  watch_today     TEXT NOT NULL DEFAULT '',      -- AI one-sentence forward-looking note
  generated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  model           TEXT NOT NULL DEFAULT ''       -- e.g. "claude-3-5-haiku-20241022"
);

-- Fast date-descending lookup (frontend always fetches most recent row ≤ today)
CREATE INDEX IF NOT EXISTS daily_briefs_date_idx ON daily_briefs (date DESC);

-- Row-level security: the edge function writes via service role, the frontend reads publicly
ALTER TABLE daily_briefs ENABLE ROW LEVEL SECURITY;

-- Public read: all users (authenticated and anonymous) can read briefs
CREATE POLICY "public read" ON daily_briefs
  FOR SELECT USING (true);

-- Service role write: only the edge function (service role key) can insert/update
CREATE POLICY "service_role_write" ON daily_briefs
  FOR ALL
  USING (auth.role() = 'service_role')
  WITH CHECK (auth.role() = 'service_role');

COMMENT ON TABLE daily_briefs IS
  'AI-generated daily market brief. One row per trading day. Generated by generate-daily-brief edge function at 6am ET via pg_cron.';
```

```typescript
// ── Helpers ───────────────────────────────────────────────────────────────────

function todayDate(): string {
  // Returns YYYY-MM-DD in US Eastern time (UTC-5 / UTC-4 DST)
  // At 6am ET the UTC offset is -4 (EDT) or -5 (EST).
  // We use UTC-4 as a safe default — worst case it's one day off on the ST/DST boundary,
  // which the retry at 6:15 would catch anyway.
  const d = new Date();
  const etOffset = 4 * 60; // EDT = UTC-4
  const etMs = d.getTime() - etOffset * 60 * 1000;
  const et = new Date(etMs);
  const y = et.getUTCFullYear();
  const m = String(et.getUTCMonth() + 1).padStart(2, "0");
  const day = String(et.getUTCDate()).padStart(2, "0");
  return `${y}-${m}-${day}`;
}

function clamp(v: number, min: number, max: number): number {
  return Math.max(min, Math.min(max, v));
}

/**
 * Compute the breadth-based sentiment score (0–100).
 * Reuses the same formula as the Trading Plan's Market Pulse widget.
 *   score = 50% × advancers_ratio + 30% × new_highs_ratio + 20% × median_norm
 */
function computeSentimentScore(
  advancing: number,
  declining: number,
  newHighs: number,
  newLows: number,
  medianReturn: number,
): number {
  const total = advancing + declining;
  if (total === 0) return 50;
  const advRatio = advancing / total;
  const hlTotal = newHighs + newLows;
  const hlRatio = hlTotal > 0 ? newHighs / hlTotal : 0.5;
  const medNorm = clamp((medianReturn + 3) / 6, 0, 1);
  return Math.round((0.5 * advRatio + 0.3 * hlRatio + 0.2 * medNorm) * 100);
}

function sentimentLabel(score: number): string {
  if (score >= 75) return "Extreme Greed";
  if (score >= 56) return "Greed";
  if (score >= 46) return "Neutral";
  if (score >= 26) return "Fear";
  return "Extreme Fear";
}

/**
 * Classify market regime from breadth + VIX data.
 */
function classifyRegime(
  advancing: number,
  declining: number,
  medianReturn: number,
  vix: number,
): string {
  const total = advancing + declining;
  const advRatio = total > 0 ? advancing / total : 0.5;
  if (Math.abs(medianReturn) > 1.5 || vix > 25) return "volatile";
  if (advRatio < 0.40 || medianReturn < -1) return "risk_off";
  if (advRatio > 0.60 && medianReturn > 0) return "trending_up";
  if (advRatio >= 0.45 && advRatio <= 0.55 && Math.abs(medianReturn) < 0.5) return "range_bound";
  // Default to trending_up when positive, range_bound otherwise
  return medianReturn > 0 ? "trending_up" : "range_bound";
}

// ── Data fetchers ─────────────────────────────────────────────────────────────

interface BreadthData {
  advancing: number;
  declining: number;
  newHighs: number;
  newLows: number;
  medianReturn: number;
}

/**
 * Fetch EODHD bulk EOD data for the previous day and compute breadth stats.
 * The bulk endpoint returns all US stocks — we walk the rows once.
 */
async function fetchBreadth(eodhKey: string): Promise<BreadthData | null> {
  try {
    const url = `${EODHD_BASE}/eod-bulk-last-day/US?api_token=${eodhKey}&fmt=json`;
    const res = await fetch(url, { signal: AbortSignal.timeout(15_000) });
    if (!res.ok) {
      console.error(`EODHD bulk: HTTP ${res.status}`);
      return null;
    }
    const rows: Array<{ change_p?: number; close?: number; adjusted_close?: number }> = await res.json();
    if (!Array.isArray(rows) || rows.length === 0) return null;

    let advancing = 0, declining = 0, newHighs = 0, newLows = 0;
    const returns: number[] = [];

    for (const row of rows) {
      const chgP = row.change_p ?? 0;
      if (chgP > 0) advancing++;
      else if (chgP < 0) declining++;
      returns.push(chgP);
      // EODHD doesn't include 52w H/L in bulk-last-day, so we estimate via
      // the daily return magnitude as a proxy for new highs/lows.
      // A return > +4% on the day is a rough proxy for a 52-week high candidate.
      if (chgP > 4) newHighs++;
      if (chgP < -4) newLows++;
    }

    returns.sort((a, b) => a - b);
    const mid = Math.floor(returns.length / 2);
    const medianReturn = returns.length % 2 === 0
      ? (returns[mid - 1] + returns[mid]) / 2
      : returns[mid];

    return { advancing, declining, newHighs, newLows, medianReturn };
  } catch (e) {
    console.error("fetchBreadth error:", e);
    return null;
  }
}

interface QuoteResult {
  symbol: string;
  changePercent: number | null;
  price: number | null;
}

/**
 * Fetch batch quotes from the api-yahoo sibling function.
 * Uses the ?endpoint=quotes&symbols= batch endpoint that returns up to 50 symbols.
 */
async function fetchYahooQuotes(
  symbols: string[],
  supabaseUrl: string,
  serviceKey: string,
): Promise<QuoteResult[]> {
  try {
    const url = `${supabaseUrl}/functions/v1/api-yahoo?endpoint=quotes&symbols=${symbols.join(",")}`;
    const res = await fetch(url, {
      headers: {
        Authorization: `Bearer ${serviceKey}`,
        apikey: serviceKey,
      },
      signal: AbortSignal.timeout(12_000),
    });
    if (!res.ok) {
      console.error(`api-yahoo quotes: HTTP ${res.status}`);
      return symbols.map((s) => ({ symbol: s, changePercent: null, price: null }));
    }
    const data: Record<string, { regularMarketChangePercent?: number | null; regularMarketPrice?: number | null } | null> = await res.json();
    return symbols.map((s) => ({
      symbol: s,
      changePercent: data[s]?.regularMarketChangePercent ?? null,
      price: data[s]?.regularMarketPrice ?? null,
    }));
  } catch (e) {
    console.error("fetchYahooQuotes error:", e);
    return symbols.map((s) => ({ symbol: s, changePercent: null, price: null }));
  }
}

interface NewsHeadline {

/**
 * Fetch top general market headlines from Finnhub.

async function fetchFinnhubNews(finnhubKey: string, limit = 15): Promise<NewsHeadline[]> {
  try {
    const url = `${FINNHUB_BASE}/news?category=general&token=${finnhubKey}`;
    const res = await fetch(url, { signal: AbortSignal.timeout(8_000) });
    if (!res.ok) return [];
    const items: Array<{ headline?: string; source?: string }> = await res.json();
    return items
      .slice(0, limit)
      .map((i) => ({ title: i.headline ?? "", source: i.source ?? "" }))
      .filter((i) => i.title.length > 0);

    console.error("fetchFinnhubNews error:", e);
    return [];
  }
}

interface ClaudeProseFields {
  summary: string;
  fear_greed_interpretation: string;
  current_events: string;
  watch_today: string;
}

/**
 * Call the Claude API once with all computed data and receive the four prose fields.
 * Claude ONLY writes prose — all numeric data comes from our own computations.
 */
async function callClaude(
  anthropicKey: string,
  data: {
    date: string;
    regime: string;
    sentimentScore: number;
    sentimentLabel: string;
    vix: number | null;
    advancing: number;
    declining: number;
    medianReturn: number;
    sectors: Array<{ name: string; symbol: string; pct_change: number | null }>;
    headlines: NewsHeadline[];
  },
): Promise<ClaudeProseFields> {
  const fallback: ClaudeProseFields = {
    summary: "",
    fear_greed_interpretation: "",
    current_events: "",
    watch_today: "",
  };

  const systemPrompt = `You are a professional market analyst writing a pre-market briefing for retail traders.
You will receive structured market data as JSON. Write the briefing by filling in the output schema below.
Populate only the four prose fields. Do not add or remove any fields.

Tone: clear, direct, factual. No hype, no price predictions, no "should" language.
Limits: max 4 sentences for summary; max 1 sentence for fear_greed_interpretation; max 3 sentences for current_events; max 1 sentence for watch_today.

Return ONLY valid JSON matching this schema:
{
  "summary": "...",
  "fear_greed_interpretation": "...",
  "current_events": "...",
  "watch_today": "..."
}`;

  const userContent = JSON.stringify({
    date: data.date,
    market_regime: data.regime,
    breadth: {
      advancing: data.advancing,
      declining: data.declining,
      advancing_pct: Math.round(data.advancing / (data.advancing + data.declining) * 100),
      median_return_pct: data.medianReturn.toFixed(2),
    },
    sentiment: { score: data.sentimentScore, label: data.sentimentLabel },
    vix: data.vix,
    sectors: data.sectors.map((s) => ({
      name: s.name,
      pct_change: s.pct_change != null ? `${s.pct_change >= 0 ? "+" : ""}${s.pct_change.toFixed(2)}%` : "N/A",
    })),
    top_headlines: data.headlines.slice(0, 10).map((h) => h.title),
  });

  // Try up to 2 times
  for (let attempt = 0; attempt < 2; attempt++) {
    try {
      const res = await fetch(CLAUDE_URL, {
        method: "POST",
        headers: {
          "x-api-key": anthropicKey,
          "anthropic-version": "2023-06-01",
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          model: CLAUDE_MODEL,
          max_tokens: 600,
          messages: [{ role: "user", content: userContent }],
          system: systemPrompt,
        }),
        signal: AbortSignal.timeout(20_000),
      });

      if (!res.ok) {
        console.error(`Claude API: HTTP ${res.status} on attempt ${attempt + 1}`);
        if (attempt === 0) { await new Promise((r) => setTimeout(r, 3000)); continue; }
        return fallback;
      }

      const body = await res.json();
      const text: string = body?.content?.[0]?.text ?? "";

      // Extract JSON from the response (strip any markdown fences if present)
      const jsonMatch = text.match(/\{[\s\S]*\}/);
      if (!jsonMatch) {
        console.error("Claude response has no JSON block");
        return fallback;
      }
      const parsed = JSON.parse(jsonMatch[0]);
      return {
        summary:                  String(parsed.summary ?? ""),
        fear_greed_interpretation: String(parsed.fear_greed_interpretation ?? ""),
        current_events:           String(parsed.current_events ?? ""),
        watch_today:              String(parsed.watch_today ?? ""),
      };
    } catch (e) {
      console.error(`Claude API attempt ${attempt + 1} error:`, e);
      if (attempt === 0) await new Promise((r) => setTimeout(r, 3000));
    }
  }
  return fallback;
}

// ── Main handler ──────────────────────────────────────────────────────────────

serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  const SUPABASE_URL  = Deno.env.get("SUPABASE_URL") ?? "";
  const SERVICE_KEY   = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";
  const EODHD_KEY     = Deno.env.get("EODHD_API_KEY") ?? "";
  const FINNHUB_KEY   = Deno.env.get("FINNHUB_API_KEY") ?? "d6hubv9r01qr5k4dbeo0d6hubv9r01qr5k4dbeog";
  const ANTHROPIC_KEY = Deno.env.get("ANTHROPIC_API_KEY") ?? "";

  if (!EODHD_KEY) {
    console.error("EODHD_API_KEY not set");
    return new Response(JSON.stringify({ error: "EODHD_API_KEY not set" }), {
      status: 500,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  const supabase = createClient(SUPABASE_URL, SERVICE_KEY);
  const today = todayDate();

  // ── 1. Idempotency check ───────────────────────────────────────────────────
  const { data: existing } = await supabase
    .from("daily_briefs")
    .select("id")
    .eq("date", today)
    .maybeSingle();

  if (existing) {
    console.log(`daily_briefs: row already exists for ${today} — skipping`);
    return new Response(JSON.stringify({ ok: true, skipped: true, date: today }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  // ── 2. Parallel data fetch ─────────────────────────────────────────────────
  const sectorSymbols = SPDR_ETFS.map((e) => e.symbol);
  const allYahooSymbols = [...INDEX_SYMBOLS, ...sectorSymbols];

  const [breadth, yahooQuotes, headlines] = await Promise.all([
    fetchBreadth(EODHD_KEY),
    fetchYahooQuotes(allYahooSymbols, SUPABASE_URL, SERVICE_KEY),
    fetchFinnhubNews(FINNHUB_KEY, 15),
  ]);

  if (!breadth) {
    console.error("Breadth data unavailable — aborting (retry will pick this up)");
    return new Response(JSON.stringify({ error: "breadth data unavailable" }), {
      status: 503,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  // ── 3. Compute derived values ──────────────────────────────────────────────

  const quoteMap = Object.fromEntries(
    yahooQuotes.map((q) => [q.symbol, q])
  );

  // VIX from Yahoo Finance
  const vixQuote = quoteMap["^VIX"];
  const vix: number | null = vixQuote?.price ?? null;

  // Sentiment score and regime
  const sentScore = computeSentimentScore(
    breadth.advancing,
    breadth.declining,
    breadth.newHighs,
    breadth.newLows,
    breadth.medianReturn,
  );
  const sentLabel = sentimentLabel(sentScore);
  const regime = classifyRegime(
    breadth.advancing,
    breadth.declining,
    breadth.medianReturn,
    vix ?? 20,
  );

  // Sector list — sorted descending by pct_change for display
  const sectors = SPDR_ETFS.map((etf) => ({
    symbol: etf.symbol,
    name: etf.name,
    pct_change: quoteMap[etf.symbol]?.changePercent ?? null,
  })).sort((a, b) => {
    if (a.pct_change == null) return 1;
    if (b.pct_change == null) return -1;
    return b.pct_change - a.pct_change;
  });

  // fear_greed object (numeric fields — Claude fills interpretation)
  const fearGreedBase = {
    score: sentScore,
    label: sentLabel,
    vix: vix,
    interpretation: "", // filled by Claude below
  };

  // headline for the collapsed card state (generated from data, not AI)
  const topSector = sectors.find((s) => s.pct_change != null && s.pct_change > 0);
  const botSector = [...sectors].reverse().find((s) => s.pct_change != null && s.pct_change < 0);
  const advPct = Math.round(breadth.advancing / (breadth.advancing + breadth.declining) * 100);
  const spxChange = quoteMap["^GSPC"]?.changePercent;
  const spxStr = spxChange != null
    ? `S&P 500 ${spxChange >= 0 ? "+" : ""}${spxChange.toFixed(2)}%`
    : "Markets";
  const dataHeadline = topSector && botSector
    ? `${spxStr} · ${topSector.name} led, ${botSector.name} lagged · ${advPct}% advanced`
    : `${spxStr} · ${advPct}% of stocks advanced`;

  // ── 4. Call Claude for prose fields ───────────────────────────────────────
  let prose: ClaudeProseFields = {
    summary: "", fear_greed_interpretation: "", current_events: "", watch_today: "",
  };

  if (ANTHROPIC_KEY) {
    prose = await callClaude(ANTHROPIC_KEY, {
      date: today,
      regime,
      sentimentScore: sentScore,
      sentimentLabel: sentLabel,
      vix,
      advancing: breadth.advancing,
      declining: breadth.declining,
      medianReturn: breadth.medianReturn,
      sectors,
      headlines,
    });
  } else {
    console.warn("ANTHROPIC_API_KEY not set — skipping AI prose, writing data-only row");
  }

  // ── 5. Upsert to daily_briefs ──────────────────────────────────────────────
  const row = {
    date: today,

    market_regime: regime,
    sentiment_score: sentScore,
    vix,
    summary: prose.summary,
    sectors: sectors,
    fear_greed: { ...fearGreedBase, interpretation: prose.fear_greed_interpretation },
    current_events: prose.current_events,
    calendar: [], // economic calendar placeholder — extend in a future task
    watch_today: prose.watch_today,
    model: ANTHROPIC_KEY ? CLAUDE_MODEL : "none",
  };

  const { error: upsertErr } = await supabase
    .from("daily_briefs")
    .upsert(row, { onConflict: "date" });

  if (upsertErr) {
    console.error("Upsert error:", upsertErr);
    return new Response(JSON.stringify({ error: upsertErr.message }), {
      status: 500,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  console.log(`daily_briefs: upserted row for ${today} (regime=${regime}, sentiment=${sentScore})`);
  return new Response(JSON.stringify({ ok: true, date: today, regime, sentiment: sentScore }), {
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });
});
```

```typescript
import { useQuery } from '@tanstack/react-query';
import { supabase } from '@/integrations/supabase/client';
import { format } from 'date-fns';

// ── Types ─────────────────────────────────────────────────────────────────────

export interface DailyBriefSector {
  symbol: string;
  name: string;
  pct_change: number | null;
}

export interface DailyBriefFearGreed {
  score: number;
  label: string;
  vix: number | null;
  interpretation: string;
}

export interface DailyBriefCalendarItem {
  time: string;
  name: string;
  estimate?: string;
  prior?: string;
}

export type MarketRegime = 'trending_up' | 'range_bound' | 'volatile' | 'risk_off';

export interface DailyBrief {
  id: string;
  date: string;                           // "YYYY-MM-DD"
  headline: string;
  market_regime: MarketRegime;
  sentiment_score: number;                // 0–100
  vix: number | null;
  summary: string;
  sectors: DailyBriefSector[];
  fear_greed: DailyBriefFearGreed;
  current_events: string;
  calendar: DailyBriefCalendarItem[];
  watch_today: string;
  generated_at: string;
  model: string;
}

// ── Hook ──────────────────────────────────────────────────────────────────────

/**
 * Fetches the most recent daily brief row with date ≤ today.
 * On weekends and holidays this returns Friday's brief, which the
 * DailyBriefCard marks with a "Previous session" badge.
 *
 * staleTime: 1 hour — the brief doesn't change during the day.
 */
export function useDailyBrief() {
  return useQuery<DailyBrief | null>({
    queryKey: ['daily-brief'],
    queryFn: async () => {
      const today = format(new Date(), 'yyyy-MM-dd');
      const { data, error } = await supabase
        .from('daily_briefs')
        .select('*')
        .lte('date', today)          // today or earlier — handles weekends/holidays
        .order('date', { ascending: false })
        .limit(1)
        .maybeSingle();              // returns null (not throws) when no rows exist

      if (error) {
        console.error('useDailyBrief error:', error);
        throw error;
      }
      return data as DailyBrief | null;
    },
    staleTime: 60 * 60_000,          // 1 hour — brief doesn't change intraday
    gcTime:    4 * 60 * 60_000,      // keep in cache 4 hours
    refetchOnWindowFocus: false,
    retry: 1,
  });
}
```
